# Login view (authentification/views.py)
@csrf_exempt
@require_POST
def login_api(request):
    try:
        data = json.loads(request.body)
    except json.JSONDecodeError:
        return JsonResponse({"detail": "JSON invalide."}, status=400)
    username = data.get("username")
    password = data.get("password")
    if not username or not password:
        return JsonResponse({"detail": "Identifiant et mot de passe requis."}, status=400)
    user = authenticate(request, username=username, password=password)
    if user is None:
        return JsonResponse({"detail": "Identifiants invalides."}, status=401)
    login(request, user)
    return JsonResponse(UserSerializer(user).data)

@deconstructible
class Telephone2Validator:
    message = "Le numéro doit contenir exactement 8 chiffres."
    def __call__(self, value):
        if value is None or value == "":
            return
        RegexValidator(regex=r"^\d{8}$", message=self.message)(value)
    def __eq__(self, other):
        return isinstance(other, Telephone2Validator)

from rest_framework.permissions import SAFE_METHODS, BasePermission
from .models import User

PERM_FINANCIAL_REPORTS = "authentification.view_financial_reports"
PERM_AUDIT_LOGS = "authentification.view_audit_logs"


def can_view_financial_reports(user) -> bool:
    return bool(user and user.is_authenticated and user.has_perm(PERM_FINANCIAL_REPORTS))

def can_view_audit_logs(user) -> bool:
    return bool(user and user.is_authenticated and user.has_perm(PERM_AUDIT_LOGS))


def effective_role(user):
    """Effective business role — superusers are treated as ADMIN."""
    if not user or not getattr(user, "is_authenticated", False):
        return None
    if getattr(user, "is_superuser", False):
        return User.Role.ADMIN
    return getattr(user, "role", None)


class IsSuperAdminRole(BasePermission):
    """Audit log access: superuser, view_audit_logs perm, or ADMIN role."""
    def has_permission(self, request, view):
        u = request.user
        if not u or not getattr(u, "is_authenticated", False):
            return False
        if getattr(u, "is_superuser", False):
            return True
        if can_view_audit_logs(u):
            return True
        return getattr(u, "role", None) == User.Role.ADMIN


class IsAdminRole(BasePermission):
    """Create / update / delete reserved to ADMIN role (or superuser)."""
    def has_permission(self, request, view):
        return effective_role(request.user) == User.Role.ADMIN


class DenyPointageAgent(BasePermission):
    """POINTAGE_AGENT can ONLY access attendance views."""
    message = "Accès réservé : ce rôle ne gère que la feuille de présence."
    def has_permission(self, request, view):
        if not request.user.is_authenticated:
            return False
        return effective_role(request.user) != User.Role.POINTAGE_AGENT


class IsAdminOrComptableFinancial(BasePermission):
    """Payments, cash register, advances: ADMIN or COMPTABLE only (no read for POINTAGE_AGENT)."""
    def has_permission(self, request, view):
        if not request.user.is_authenticated:
            return False
        r = effective_role(request.user)
        return r in (User.Role.ADMIN, User.Role.COMPTABLE)


class IsAdminOrPointageAgentWrite(BasePermission):
    """Read: any authenticated user. Write: ADMIN or POINTAGE_AGENT."""
    def has_permission(self, request, view):
        if not request.user.is_authenticated:
            return False
        if request.method in SAFE_METHODS:
            return True
        r = effective_role(request.user)
        return r in (User.Role.ADMIN, User.Role.POINTAGE_AGENT)


class IsAdminOrComptable(BasePermission):
    """Accounting actions (salaries, expenses): ADMIN or COMPTABLE."""
    def has_permission(self, request, view):
        if not request.user.is_authenticated:
            return False
        r = effective_role(request.user)
        return r in (User.Role.ADMIN, User.Role.COMPTABLE)


# Function-based view decorators (authentification/decorators.py)

    return permission_required(PERM_FINANCIAL_REPORTS, raise_exception=True)(view_func)

def audit_logs_required(view_func):
    return permission_required(PERM_AUDIT_LOGS, raise_exception=True)(view_func)

    password = serializers.CharField(write_only=True, required=False, allow_blank=True)

    role_label = serializers.CharField(source="get_role_display", read_only=True)

    class Meta:
        model = User
        fields = ("id","username","email","first_name","last_name",
                  "display_name","role","role_label","password","generate_password")
        read_only_fields = ("id",)

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._plain_password_to_return = None

    def validate_role(self, value):
        allowed = (User.Role.ADMIN, User.Role.POINTAGE_AGENT, User.Role.COMPTABLE)
        if value not in allowed:
            raise serializers.ValidationError("Rôle non autorisé pour ce formulaire.")
        return value

    def validate(self, attrs):
        is_create = self.instance is None
        gen = bool(attrs.get("generate_password"))
        pwd = attrs.get("password")

        if is_create:
            if gen:
                plain = secrets.token_urlsafe(16)
                attrs["password"] = plain
                self._plain_password_to_return = plain
            elif not pwd or not str(pwd).strip():
                raise serializers.ValidationError(
                    {"password": "Indiquez un mot de passe ou cochez la génération automatique."}
                )
        else:
            if gen:
                plain = secrets.token_urlsafe(16)
                attrs["password"] = plain
                self._plain_password_to_return = plain
            elif pwd is not None and not str(pwd).strip():
                attrs.pop("password", None)

        attrs.pop("generate_password", None)
        return attrs

    def create(self, validated_data):
        password = validated_data.pop("password")
        user = User(**validated_data)
        user.set_password(password)
        user.save()
        return user

    def update(self, instance, validated_data):
        password = validated_data.pop("password", None)
        for key, value in validated_data.items():
            setattr(instance, key, value)
        if password:
            instance.set_password(password)
        instance.save()
        return instance

@action(detail=True, methods=["post"], url_path="executer_paiement")
def executer_paiement(self, request, pk=None):
    """
    Records a net payment: pointages FIFO, advances absorbed on oldest brut,
    advance marked as solde when fully covered, accounting Depense written.
    Body JSON: { "montant": "123.45", "date_paiement": "YYYY-MM-DD" (optional) }
    """
    prof = self.get_object()
    today = timezone.localdate()
    body = request.data if isinstance(request.data, dict) else {}
    raw = body.get("montant")
    if raw is None:
        return Response({"montant": "Le champ montant est requis."},
                        status=status.HTTP_400_BAD_REQUEST)
    try:
        montant_net = Decimal(str(raw).strip().replace(",", "."))
    except Exception:
        return Response({"montant": "Montant invalide."}, status=status.HTTP_400_BAD_REQUEST)

    date_raw = body.get("date_paiement") or body.get("date")
    if date_raw not in (None, ""):
        from django.utils.dateparse import parse_date
        date_depense = parse_date(str(date_raw))
        if date_depense is None:
            return Response({"date_paiement": "Date invalide (attendu AAAA-MM-JJ)."},
                            status=status.HTTP_400_BAD_REQUEST)
    else:
        date_depense = today

    raw_per = body.get("perimetre")
    if raw_per in (None, ""):
        perimetre_paiement = None
    elif isinstance(raw_per, str) and raw_per.strip() == "mois_courant":
        perimetre_paiement = "mois_courant"
    else:
        return Response({"perimetre": "Valeur attendue : mois_courant ou omis."},
                        status=status.HTTP_400_BAD_REQUEST)

    try:
        out = _executer_paiement_professeur(
            prof, montant_net, date_depense, today, perimetre=perimetre_paiement
        )
    except ValidationError as e:
        return Response(e.detail, status=status.HTTP_400_BAD_REQUEST)
    return Response(out, status=status.HTTP_200_OK)

        fields = ("id","eleve","eleve_nom","eleve_prenom","eleve_classe","type_frais",
                  "montant_du","montant_paye","reste_a_payer","eleve_reste_a_payer_annuel",
                  "eleve_snapshot","date_encaissement","mois_concerne","mois_concernes",
                  "methode","remarque","progression_annuelle")

    def validate_mois_concernes(self, value):
        if value is None:
            return []
        if not isinstance(value, list):
            raise serializers.ValidationError("Attendu : une liste de codes mois (ex. [\"09\", \"10\"]).")
        allowed = set(_ACADEMIC_MONTH_ORDER)
        out, seen = [], set()
        for item in value:
            s = str(item).strip().zfill(2)
            if s not in allowed:
                raise serializers.ValidationError(f"Mois non reconnu pour l'année scolaire : {item!r}.")
            if s not in seen:
                seen.add(s)
                out.append(s)
        return out

    def validate(self, attrs):
        mois = attrs.get("mois_concernes")
        if self.instance is None:
            if not mois:
                raise serializers.ValidationError(
                    {"mois_concernes": "Indiquez au moins un mois couvert par cet encaissement."}
                )
        elif "mois_concernes" in attrs and not mois:
            raise serializers.ValidationError(
                {"mois_concernes": "La liste des mois ne peut pas être vide."}
            )

        eleve = attrs.get("eleve")
        if eleve is None and self.instance is not None:
            eleve = self.instance.eleve
        montant_du = attrs.get("montant_du")
        if montant_du is None and self.instance is not None:
            montant_du = self.instance.montant_du
        if mois is None and self.instance is not None:
            mois = self.instance.mois_concernes or []

        # Server-side cross-check: the declared `montant_du` must match the
        # expected outstanding for the selected months (tolerance 0.02 DT).
        if eleve is not None and mois and montant_du is not None:
            exclude_id = self.instance.pk if self.instance else None
            exp = expected_montant_du_for_mois(eleve, mois, exclude_paiement_id=exclude_id)
            md = montant_du if isinstance(montant_du, Decimal) else Decimal(str(montant_du))
            if (md - exp).copy_abs() > Decimal("0.02"):
                raise serializers.ValidationError(
                    {"montant_du": (
                        f"Pour les mois sélectionnés, le reste à payer attendu sur l’échéance est {exp} DT "
                        f"(selon la répartition actuelle des encaissements)."
                    )}
                )
        return attrs

    def create(self, validated_data):
        instance = super().create(validated_data)
        # FIFO re-allocation of mois_concernes against effective amount paid.
        codes = fifo_mois_concernes_for_payment(
            instance.eleve, instance.montant_paye, exclude_paiement_id=instance.pk
        )
        if codes != instance.mois_concernes:
            instance.mois_concernes = codes
            instance.save(update_fields=["mois_concernes"])
        return instance

    def update(self, instance, validated_data):
        instance = super().update(instance, validated_data)
        codes = fifo_mois_concernes_for_payment(
            instance.eleve, instance.montant_paye, exclude_paiement_id=instance.pk
        )
        if codes != instance.mois_concernes:
            instance.mois_concernes = codes
            instance.save(update_fields=["mois_concernes"])
        return instance

def audit_user(request):
    """Authenticated user for the request, or None."""
    if request is None:
        return None
    user = getattr(request, "user", None)

import { STORAGE_AUTH_KEY } from "./config.js";

export const ROLE = {
  ADMIN: "ADMIN",
  POINTAGE_AGENT: "POINTAGE_AGENT",
  COMPTABLE: "COMPTABLE",
};

export function readStoredUserRaw() {
  try {
    const raw = localStorage.getItem(STORAGE_AUTH_KEY);
    if (!raw) return null;
    return JSON.parse(raw);
  } catch {
    return null;
  }
}

export function userPayloadFromApi(data) {
  if (!data || typeof data !== "object") return null;
  return {
    username: data.username ?? "",
    display_name: data.display_name ?? "",
    first_name: data.first_name ?? "",
    last_name: data.last_name ?? "",
    email: data.email ?? "",
    role: data.role ?? "",
    role_label: data.role_label ?? "",
    is_superuser: Boolean(data.is_superuser),
  };
}

export function isAdminUser(u) {
  if (!u) return false;
  if (u.is_superuser) return true;
  return u.role === ROLE.ADMIN;
}

export function isPointageAgent(u) {
  return Boolean(u && u.role === ROLE.POINTAGE_AGENT);
}
export function isComptable(u) {
  return Boolean(u && u.role === ROLE.COMPTABLE);
}

export const POINTAGE_AGENT_HOME = "/pointage-personnel";

/** Routes allowed for COMPTABLE role only (no ADMIN, no superuser). */
export const COMPTABLE_ALLOWED_PATHS = ["/caisse", "/paiements", "/depenses"];

export function isAdminOrComptable(u) {
  return isAdminUser(u) || isComptable(u);
}
export function isPointageOnlyAgent(u) {
  return isPointageAgent(u) && !isAdminUser(u);
}

/**
 * After login or on access denied:
 *   POINTAGE_AGENT  -> /pointage-personnel
 *   COMPTABLE (non admin) -> /caisse
 *   ADMIN / superuser -> /dashboard
 */
export function defaultHomePath(u) {
  if (!u) return "/dashboard";
  if (u.role === ROLE.POINTAGE_AGENT && !isAdminUser(u)) return "/pointage-personnel";
  if (isComptable(u) && !isAdminUser(u)) return "/caisse";
  return "/dashboard";
}

export function displayUserName(u) {
  if (!u) return "";
  const d = typeof u.display_name === "string" ? u.display_name.trim() : "";
  if (d) return d;
  const parts = [u.first_name, u.last_name].filter(Boolean);
  if (parts.length) return parts.join(" ");
  return u.username || "";
}

/**
 * Path access control (mirrors sidebar):
 *   /dashboard               -> admins only
 *   POINTAGE_AGENT (non admin) -> /pointage-personnel only
 *   COMPTABLE (non admin)    -> /caisse, /paiements, /depenses
 *   ADMIN / superuser        -> all modules
 */
export function canAccessPath(pathname, u) {
  if (!u || !u.role) return false;
  if (pathname === "/dashboard") {
    return isAdminUser(u);
  }
  if (u.role === ROLE.POINTAGE_AGENT && !isAdminUser(u)) {
    return pathname === "/pointage-personnel";
  }
  if (
    pathname === "/audit-logs" ||
    pathname === "/settings" ||
    pathname === "/emploi-du-temps" ||
    pathname === "/historique-emploi" ||
    pathname === "/gestion-classes" ||
    pathname === "/personnel" ||
    pathname === "/pointage-personnel"
  ) {
    return isAdminUser(u);
  }
  if (isAdminUser(u)) return true;
  if (u.role === ROLE.COMPTABLE) {
    return COMPTABLE_ALLOWED_PATHS.includes(pathname);
  }
  return false;
}

  const { pathname } = useLocation();
  if (!user) return <Navigate to="/login" replace />;
  if (!canAccessPath(pathname, user)) return <Navigate to={defaultHomePath(user)} replace />;
  return children;
}

// AuthContext.jsx — initial state read from localStorage, refresh from /api/auth/me/
import { createContext, useCallback, useContext, useMemo, useState } from "react";
import { API_AUTH_ME, STORAGE_AUTH_KEY, apiFetch } from "./config.js";
import { readStoredUserRaw, userPayloadFromApi } from "./authRoles.js";

export function AuthProvider({ children }) {
  const [user, setUser] = useState(() => readStoredUserRaw());
  const refreshUser = useCallback(async () => {
    try {
      const res = await apiFetch(API_AUTH_ME);
      if (!res.ok) return;
      const data = await res.json();
      const payload = userPayloadFromApi(data);
      if (payload) {
        localStorage.setItem(STORAGE_AUTH_KEY, JSON.stringify(payload));
        setUser(payload);
      }
    } catch { /* session expirée ou réseau */ }
  }, []);
  // ...
}

// config.js — fetch wrapper. Note: `accessToken` (Bearer) is supported on the
// client but the backend only honours session cookies; no JWT is issued.
export const STORAGE_AUTH_KEY = "erp_auth";
export function getAuthHeaders() {
  const headers = { Accept: "application/json" };
  try {
    const raw = localStorage.getItem(STORAGE_AUTH_KEY);
    if (raw) {
      const u = JSON.parse(raw);
      if (u.accessToken && typeof u.accessToken === "string") {
        headers.Authorization = `Bearer ${u.accessToken}`;
      }
    }
  } catch { /* ignore */ }
  return headers;
}
export function apiFetch(url, options = {}) {
  return fetch(url, {
    credentials: "include",
    ...options,
    headers: { ...getAuthHeaders(), ...options.headers },
  });
}

--- KNOWN GOTCHAS / POINTS WORTH AUDITING ---
================================================================================
1.  DEBUG = True is still set in core/settings.py; ALLOWED_HOSTS is empty
    when DEBUG is False. SECRET_KEY in code is a "django-insecure-..." dev
    placeholder ([HIDDEN] in this brief).
2.  Cookies have SECURE = False and SAMESITE = "Lax" — fine for dev over
    HTTP, MUST be reviewed for production (HTTPS only, Secure=True).
3.  CSRF protection is DISABLED on every DRF endpoint via
    `SessionAuthenticationWithoutCSRF`. POST/PUT/PATCH/DELETE only require
    a valid session cookie. With CORS_ALLOW_CREDENTIALS=True and a tight
    CORS_ALLOWED_ORIGINS this is a deliberate trade-off, but it makes the
    CSRF posture entirely dependent on browser SameSite enforcement.
4.  `login_api` and `logout_api` are `@csrf_exempt` and accept JSON bodies.
    There is NO rate-limiting / lockout on failed logins. NO password
    reset / email verification flow.
5.  REST_FRAMEWORK default permission is `AllowAny`; every ViewSet must
    therefore set `permission_classes` explicitly. Any new view that
    forgets to set it is unauthenticated by default.
6.  UserManagementViewSet allows any ADMIN to delete any non-superuser
    (including other ADMINs and themselves). No confirmation gate, no
    audit log entry on user deletion (audit only covers scolarite models).
7.  StaffUserSerializer can return `generated_password` in the response
    payload on create/update — review whether this is logged client-side
    or stored anywhere in localStorage.
8.  PaiementSalaireWriteSerializer accepts `montant` verbatim. There is no
    server-side check that the amount matches Personnel.salaire_mensuel
    minus retenues (the resume_paie endpoint is informational only).
9.  PaiementSerializer.validate() bounds `montant_du` to the expected
    outstanding ±0.02, but `montant_paye` is unbounded — over-payment
    becomes a credit (acompte) on the student.
10. AuditLog entries are best-effort: if the helpers in
    scolarite/audit_logs.py fail, exceptions are swallowed/raised in-band
    with the action — the log can be missing while the action succeeded
    or failed depending on transaction boundaries.
11. Frontend role checks are advisory (UI only). All real enforcement
    must be on the backend ViewSet `permission_classes`. Ensure no
    sensitive data leaks via list endpoints that lack a queryset filter
    (e.g. `EleveViewSet` returns full list of students to any non-POINTAGE
    user — confirm this is intended).
12. `accessToken` Bearer header support exists on the client (`config.js`)
    but no backend authentication class consumes it — dead code. Remove
    or wire to TokenAuthentication consciously.
